"""
運算思維的四步驟主要包括
拆解（Decomposition）、
規則辨識（Pattern Recognition）、
抽象化（Abstraction）和
演算法設計（Algorithm）。
"""
#
import numpy as np
import matplotlib
matplotlib.use('TkAgg')
import matplotlib.pyplot as plt
#
from shapely import affinity
from shapely.geometry import LineString, Point
#
# --- 核心資料結構取代 eval() ---
# 初始化全域圓盤變數
# 這個字典會儲存所有柱子（A, B, C）及其對應編號的圓盤物件
disks_dict = {
    'A': {},
    'B': {},
    'C': {},
    'start': {} # 用於初始顯示
}
#
def motif_fill(LS, color, ZORDER, Alpha):
    xs, ys = LS.xy
    ax.fill(xs, ys, alpha=Alpha, closed=True, edgecolor=None, facecolor=color, fill=True, zorder=ZORDER)

# --- 繪製函數優化 ---
def draw_pole_disks(pole_name):
    global my_order
    my_order += 1
    pole_list = {'A': A_pole, 'B': B_pole, 'C': C_pole}[pole_name]

    j = len(pole_list)
    for i in range(len(pole_list)):
        disk_size = pole_list[i]
        # 直接從字典取得物件，不再使用 eval()
        disk = disks_dict[pole_name][disk_size]

        t = j - 4 - i
        t_disk = affinity.translate(disk, 0, t)
        motif_fill(t_disk, disk_color[disk_size], my_order, 1.0)

def moveDisk(height, fp, tp):
    global A_pole, B_pole, C_pole, step, my_order, box, rect, step_txt_xy

    step += 1
    my_order += 1
    print(f"step {step} --> moving disk {height} from {fp} to {tp}")

    motif_fill(box, 'white', my_order, 1.0)

    for i in (0,1,2):
        xs, ys = v_line[i].xy
        ax.plot(xs, ys, color='black', alpha=1.0, linewidth=4.0, solid_capstyle='round', zorder=my_order)

    # 模擬堆疊操作
    poles = {'A': A_pole, 'B': B_pole, 'C': C_pole}
    poles[fp].pop(0)
    poles[tp].insert(0, height)

    print(f'A = {A_pole}\nB = {B_pole}\nC = {C_pole}\n')

    if A_pole: draw_pole_disks('A')
    if B_pole: draw_pole_disks('B')
    if C_pole: draw_pole_disks('C')

    ax.annotate(f"step {step}", step_txt_xy, color='blue', fontsize=18, zorder=my_order)

    filename = f'./png/step-{step:02d}.png'
    plt.savefig(filename, format="png", dpi=72)

    plt.pause(1.4)
    ax.annotate(f"step {step}", step_txt_xy, color='white', fontsize=18, zorder=my_order)
    motif_fill(rect, 'white', my_order+1, 1.0)

# --- 設定圓盤物件 ---
def setup_all_disks(num):
    # 初始左側顯示用的圓盤
    for i in range(1, num + 1):
        offset = num - i
        disks_dict['start'][i] = LineString([
            (offset, num-i), (2*num-offset, num-i),
            (2*num-offset, num-i+1), (offset, num-i+1)
        ])

    # 柱子 A, B, C 的圓盤位置定義
    for i in range(1, num + 1):
        offset = num - i
        # A 柱
        disks_dict['A'][i] = LineString([(offset, 3), (2*num-offset, 3), (2*num-offset, 4), (offset, 4)])
        # B 柱 (向右平移 2*num+1)
        shift_b = 2*num + 1
        disks_dict['B'][i] = LineString([(offset+shift_b, 3), (2*num-offset+shift_b, 3), (2*num-offset+shift_b, 4), (offset+shift_b, 4)])
        # C 柱 (再向右平移 2*num+1)
        shift_c = 2*(2*num + 1)
        disks_dict['C'][i] = LineString([(offset+shift_c, 3), (2*num-offset+shift_c, 3), (2*num-offset+shift_c, 4), (offset+shift_c, 4)])

def hanoi(height, fromPole, intermediate, toPole):
    if height >= 1:
        hanoi(height-1, fromPole, toPole, intermediate)
        moveDisk(height, fromPole, toPole)
        hanoi(height-1, intermediate, fromPole, toPole)

# --- 主程式 ---
print('main running ...\n')

num_of_disk = 3
r0 = 3*(2*num_of_disk)+2
step_txt_xy = (r0/2-2, num_of_disk+7)
start_txt_xy = (r0/2-2, num_of_disk+7)
rect = LineString([(r0/2-7, num_of_disk+6), (r0/2+5, num_of_disk+6), (r0/2+5, num_of_disk+9), (r0/2-7, num_of_disk+9)])
box = LineString([(0,0), (r0,0), (r0,2*num_of_disk), (0,2*num_of_disk), (0,0)])

v_line_M = LineString([(r0/2,0), (r0/2,num_of_disk+1)])
v_line_L = affinity.translate(v_line_M, -(num_of_disk*2+1), 0)
v_line_R = affinity.translate(v_line_M, num_of_disk*2+1, 0)
v_line = [v_line_L, v_line_M, v_line_R]

setup_all_disks(num_of_disk)
disk_color = ['zzz', 'blue', 'gold', 'green', 'magenta', 'cyan']

plt.ion()
fig = plt.figure(figsize=(8,4.8), num="Tower of Hanoi")
ax = fig.add_subplot(1, 1, 1)
plt.xlim(-0.1, r0+0.1)
plt.ylim(0, 12.5)
plt.gca().set_aspect('equal', adjustable='box')
ax.grid(False)

# 畫圓柱
for line in v_line:
    xs, ys = line.xy
    ax.plot(xs, ys, color='black', linewidth=4.0, solid_capstyle='round', zorder=5)

# 起始狀態
for i in range(1, num_of_disk+1):
    disk = disks_dict['start'][i]
    xs, ys = disk.xy
    ax.fill(xs, ys, facecolor=disk_color[i], zorder=10)

ax.annotate("start", start_txt_xy, color='blue', fontsize=20, zorder=10)
plt.savefig('./png/step-00.png', format="png", dpi=72)
plt.pause(1.4)
motif_fill(rect, 'white', 11, 1.0)

A_pole = list(range(1, num_of_disk+1))
B_pole, C_pole = [], []
my_order, step = 30, 0

hanoi(num_of_disk, "A", "B", "C")

ax.annotate(f"step {step}", step_txt_xy, color='blue', fontsize=18, zorder=my_order+1)
print('done')
plt.pause(2)
